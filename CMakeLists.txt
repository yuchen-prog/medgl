cmake_minimum_required(VERSION 3.25)

project(MedGL VERSION 0.1.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "MedGL v1 currently requires Windows x64.")
endif()

option(MEDGL_ENABLE_CUDA "Build CUDA components; OFF is host validation only" ON)
option(MEDGL_BUILD_APP "Build DICOM reader, codecs, CLI and B/S server" OFF)

add_library(medgl_build_options INTERFACE)
target_compile_features(medgl_build_options INTERFACE cxx_std_17)
target_compile_definitions(medgl_build_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(medgl_build_options INTERFACE
        $<$<COMPILE_LANGUAGE:CXX>:/W4>
        $<$<COMPILE_LANGUAGE:CXX>:/WX>
        $<$<COMPILE_LANGUAGE:CXX>:/permissive->
    )
endif()

add_library(medgl_foundation STATIC src/core/data_contracts.cpp src/core/geometry.cpp)
target_include_directories(medgl_foundation PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(medgl_foundation PUBLIC medgl_build_options)

include(CTest)
if(BUILD_TESTING)
    add_executable(medgl_host_smoke tests/host_smoke.cpp)
    target_link_libraries(medgl_host_smoke PRIVATE medgl_foundation)
    add_test(NAME medgl.host.smoke COMMAND medgl_host_smoke)
    set_tests_properties(medgl.host.smoke PROPERTIES LABELS "host")
    add_executable(medgl_geometry_test tests/geometry_test.cpp)
    target_link_libraries(medgl_geometry_test PRIVATE medgl_foundation)
    add_test(NAME medgl.host.geometry COMMAND medgl_geometry_test)
    set_tests_properties(medgl.host.geometry PROPERTIES LABELS "host")
endif()

if(MEDGL_ENABLE_CUDA)
    include(CheckLanguage)
    check_language(CUDA)
    if(NOT CMAKE_CUDA_COMPILER)
        message(FATAL_ERROR
            "CUDA Toolkit was not found. Install a Toolkit compatible with MSVC, "
            "then run the CUDA preset. host-only validates C++ only, not rendering."
        )
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)

    add_library(medgl_renderer STATIC src/core/renderer.cu src/core/phantom.cpp)
    target_link_libraries(medgl_renderer PUBLIC medgl_foundation PRIVATE CUDA::cudart)
    set_target_properties(medgl_renderer PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    if(BUILD_TESTING)
        add_executable(medgl_crop_render_test tests/crop_render_test.cpp)
        target_link_libraries(medgl_crop_render_test PRIVATE medgl_renderer)
        add_test(NAME medgl.cuda.crop COMMAND medgl_crop_render_test)
        set_tests_properties(medgl.cuda.crop PROPERTIES LABELS "gpu" TIMEOUT 60)
        add_executable(medgl_renderer_test tests/renderer_test.cpp)
        target_link_libraries(medgl_renderer_test PRIVATE medgl_renderer)
        add_test(NAME medgl.cuda.renderer COMMAND medgl_renderer_test)
        set_tests_properties(medgl.cuda.renderer PROPERTIES LABELS "gpu" TIMEOUT 60)
        add_executable(medgl_preset_render_test tests/preset_render_test.cpp)
        target_include_directories(medgl_preset_render_test PRIVATE "${PROJECT_SOURCE_DIR}/src/server")
        target_link_libraries(medgl_preset_render_test PRIVATE medgl_renderer)
        add_test(NAME medgl.cuda.presets COMMAND medgl_preset_render_test)
        set_tests_properties(medgl.cuda.presets PROPERTIES LABELS "gpu" TIMEOUT 60)
        add_executable(medgl_orientation_render_test tests/orientation_render_test.cpp)
        target_link_libraries(medgl_orientation_render_test PRIVATE medgl_renderer)
        add_test(NAME medgl.cuda.orientation COMMAND medgl_orientation_render_test)
        set_tests_properties(medgl.cuda.orientation PROPERTIES LABELS "gpu" TIMEOUT 60)
    endif()

    add_executable(medgl_cuda_probe src/diagnostics/cuda_probe.cu)
    target_link_libraries(medgl_cuda_probe PRIVATE medgl_build_options CUDA::cudart)
    set_target_properties(medgl_cuda_probe PROPERTIES
        CUDA_STANDARD 17
        CUDA_STANDARD_REQUIRED ON
        CUDA_EXTENSIONS OFF
    )
    if(BUILD_TESTING)
        add_test(NAME medgl.cuda.probe COMMAND medgl_cuda_probe)
        set_tests_properties(medgl.cuda.probe PROPERTIES LABELS "gpu" TIMEOUT 60)
    endif()
else()
    message(STATUS "HOST-ONLY validation: no CUDA components or renderer are built.")
endif()

if(MEDGL_BUILD_APP)
    if(NOT MEDGL_ENABLE_CUDA)
        message(FATAL_ERROR "The application requires CUDA.")
    endif()
    find_package(GDCM CONFIG REQUIRED)
    find_package(PNG REQUIRED)
    find_package(libjpeg-turbo CONFIG REQUIRED)
    find_package(nlohmann_json CONFIG REQUIRED)
    find_package(boost_beast CONFIG REQUIRED)
    find_package(boost_asio CONFIG REQUIRED)

    add_library(medgl_io STATIC src/io/dicom.cpp src/io/image_codec.cpp)
    target_link_libraries(medgl_io PUBLIC medgl_foundation PRIVATE gdcmMSFF PNG::PNG libjpeg-turbo::turbojpeg)
    if(MSVC)
        target_compile_options(medgl_io PRIVATE /external:W0)
    endif()
    add_executable(medgl_cli src/cli/main.cpp)
    target_link_libraries(medgl_cli PRIVATE medgl_io medgl_renderer)
    add_library(medgl_dicom_audit_lib STATIC src/io/dicom_audit.cpp)
    target_link_libraries(medgl_dicom_audit_lib PUBLIC medgl_io nlohmann_json::nlohmann_json PRIVATE gdcmMSFF)
    target_compile_options(medgl_dicom_audit_lib PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/external:W0>)
    add_executable(medgl_dicom_audit src/diagnostics/dicom_audit.cpp)
    target_link_libraries(medgl_dicom_audit PRIVATE medgl_dicom_audit_lib gdcmMSFF)
    target_compile_options(medgl_dicom_audit PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/external:W0>)
    add_executable(medgl_vr_benchmark src/diagnostics/vr_benchmark.cpp)
    target_include_directories(medgl_vr_benchmark PRIVATE "${PROJECT_SOURCE_DIR}/src/server")
    target_link_libraries(medgl_vr_benchmark PRIVATE medgl_renderer medgl_io nlohmann_json::nlohmann_json CUDA::cudart PNG::PNG)
    add_executable(medgl_server src/server/main.cpp)
    target_link_libraries(medgl_server PRIVATE medgl_io medgl_renderer Boost::beast Boost::asio nlohmann_json::nlohmann_json)
    if(MSVC)
        target_compile_options(medgl_server PRIVATE /bigobj /external:W0)
    endif()
    if(BUILD_TESTING)
        add_executable(medgl_image_codec_test tests/image_codec_test.cpp)
        target_link_libraries(medgl_image_codec_test PRIVATE medgl_io medgl_renderer PNG::PNG windowscodecs ole32)
        add_test(NAME medgl.image.codec COMMAND medgl_image_codec_test)
        set_tests_properties(medgl.image.codec PROPERTIES LABELS "gpu;codec" TIMEOUT 120)
        add_executable(medgl_dicom_test tests/dicom_test.cpp)
        target_link_libraries(medgl_dicom_test PRIVATE medgl_io gdcmMSFF)
        target_compile_options(medgl_dicom_test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/external:W0>)
        add_test(NAME medgl.dicom.reader COMMAND medgl_dicom_test)
        set_tests_properties(medgl.dicom.reader PROPERTIES LABELS "dicom" TIMEOUT 120)
        add_executable(medgl_dicom_audit_test tests/dicom_audit_test.cpp)
        target_link_libraries(medgl_dicom_audit_test PRIVATE medgl_dicom_audit_lib gdcmMSFF)
        target_compile_options(medgl_dicom_audit_test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/external:W0>)
        add_test(NAME medgl.dicom.audit COMMAND medgl_dicom_audit_test $<TARGET_FILE:medgl_dicom_audit>)
        set_tests_properties(medgl.dicom.audit PROPERTIES LABELS "dicom" TIMEOUT 180)
        add_executable(medgl_benchmark_report_test tests/benchmark_report_test.cpp)
        target_link_libraries(medgl_benchmark_report_test PRIVATE medgl_io nlohmann_json::nlohmann_json)
        add_test(NAME medgl.benchmark.report COMMAND medgl_benchmark_report_test $<TARGET_FILE:medgl_vr_benchmark>)
        set_tests_properties(medgl.benchmark.report PROPERTIES LABELS "gpu;benchmark" TIMEOUT 180)
        add_executable(medgl_server_self_test src/server/self_test.cpp)
        target_compile_definitions(medgl_server_self_test PRIVATE MEDGL_SERVER_SELF_TEST)
        target_link_libraries(medgl_server_self_test PRIVATE medgl_foundation Boost::beast Boost::asio nlohmann_json::nlohmann_json)
        target_compile_options(medgl_server_self_test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/bigobj>)
        add_test(NAME medgl.server.protocol COMMAND medgl_server_self_test)
        set_tests_properties(medgl.server.protocol PROPERTIES LABELS "server" TIMEOUT 90)
    endif()
endif()
